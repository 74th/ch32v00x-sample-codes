cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_gpio_tester C CXX ASM)
pico_sdk_init()
add_executable(pico_gpio_tester main.c protocol.c gpio_fixture.c fixture.c uart_fixture.c adc_fixture.c i2c_fixture.c spi_fixture.c ssd1306_fixture.c ssd1306_model.c ssd1306_hw.c ws2812_fixture.c ws2812_classify.c ws2812_hw.c)
target_link_libraries(pico_gpio_tester pico_stdlib hardware_pwm hardware_clocks hardware_uart hardware_i2c pico_i2c_slave hardware_spi hardware_pio hardware_dma)
pico_enable_stdio_usb(pico_gpio_tester 1)
pico_enable_stdio_uart(pico_gpio_tester 0)
pico_add_extra_outputs(pico_gpio_tester)

pico_generate_pio_header(pico_gpio_tester ${CMAKE_CURRENT_LIST_DIR}/spi_loopback.pio)
pico_generate_pio_header(pico_gpio_tester ${CMAKE_CURRENT_LIST_DIR}/spi_dut_slave.pio)
pico_generate_pio_header(pico_gpio_tester ${CMAKE_CURRENT_LIST_DIR}/ws2812_decode.pio)
