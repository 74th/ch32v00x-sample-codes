cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_gpio_tester C CXX ASM)
pico_sdk_init()
add_executable(pico_gpio_tester main.c)
target_link_libraries(pico_gpio_tester pico_stdlib)
pico_enable_stdio_usb(pico_gpio_tester 1)
pico_enable_stdio_uart(pico_gpio_tester 0)
pico_add_extra_outputs(pico_gpio_tester)
